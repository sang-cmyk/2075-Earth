### 프롬프트 5 — `src/components/ChromaImg.tsx` (새 파일 만들기)

````
`src/components/ChromaImg.tsx` 파일을 새로 만들고 다른 파일은 건드리지 마. 내용은 한 글자도 바꾸지 말고 그대로 넣어줘. 끝나면 "완료"만 말해줘.

```tsx
import React, { useEffect, useState } from 'react';
import { assetUrl, type AssetName } from '../game/assets.ts';

// 캐릭터·아이콘 이미지는 연두색(#1CFB08 근처) 배경에 그려져 있다.
// 화면에 올릴 때 그 배경만 투명하게 지워서 보여 준다 (원본 파일은 그대로).
const cache = new Map<string, Promise<string>>();

function keyOutGreen(src: string): Promise<string> {
  const hit = cache.get(src);
  if (hit) return hit;
  const job = new Promise<string>((resolve) => {
    const img = new Image();
    img.onload = () => {
      try {
        const canvas = document.createElement('canvas');
        canvas.width = img.naturalWidth;
        canvas.height = img.naturalHeight;
        const ctx = canvas.getContext('2d');
        if (!ctx) return resolve(src);
        ctx.drawImage(img, 0, 0);
        const data = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const px = data.data;
        for (let i = 0; i < px.length; i += 4) {
          const r = px[i];
          const g = px[i + 1];
          const b = px[i + 2];
          const d = g - Math.max(r, b);
          if (d > 120) {
            px[i + 3] = 0;
          } else if (d > 50) {
            // 가장자리: 반투명 + 초록 번짐 제거
            px[i + 3] = Math.round(255 * (1 - (d - 50) / 70));
            px[i + 1] = Math.max(r, b);
          }
        }
        ctx.putImageData(data, 0, 0);
        resolve(canvas.toDataURL('image/png'));
      } catch {
        resolve(src);
      }
    };
    img.onerror = () => resolve(src);
    img.src = src;
  });
  cache.set(src, job);
  return job;
}

interface ChromaImgProps extends Omit<React.ImgHTMLAttributes<HTMLImageElement>, 'src'> {
  name: AssetName;
}

export const ChromaImg: React.FC<ChromaImgProps> = ({ name, className = '', alt = '', ...rest }) => {
  const [url, setUrl] = useState<string | null>(null);
  useEffect(() => {
    let alive = true;
    keyOutGreen(assetUrl(name)).then((u) => alive && setUrl(u));
    return () => {
      alive = false;
    };
  }, [name]);
  return <img {...rest} alt={alt} src={url ?? undefined} className={`${className} ${url ? '' : 'opacity-0'}`} />;
};
````

````
          <div className="p-3 bg-slate-950 rounded-xl border border-slate-800 text-xs text-slate-400 flex items-center justify-center gap-2">
            <Users className="w-4 h-4 text-emerald-400" />
            지금 우리 모둠 <strong className="text-white">{seatedCount}명</strong> 입장
          </div>
          <div className="flex items-center justify-center gap-2 text-xs text-slate-400 animate-pulse">
            <Clock className="w-4 h-4 text-emerald-400" />
            선생님이 게임을 시작할 때까지 잠시 기다려 주세요.
          </div>
        </div>
      </div>
    );
  }

  const myLocks = myCountry.locks || {};
  const isVotingPhase = roomData.phase === 'voting' || roomData.phase === 'revoting';
  const moneyHistory = myCountry.moneyHistory || [];
  const moneyDelta =
    moneyHistory.length >= 2 ? moneyHistory[moneyHistory.length - 1] - moneyHistory[moneyHistory.length - 2] : 0;

  const spend =
    CONFIG.domains.reduce((sum, dom) => sum + dom.cost[levels[dom.id]], 0) + CONFIG.appropriate.cost[appropriate];
  const remaining = Math.round(myCountry.money) - spend;

  const statusBar = (
    <div className="p-4 bg-slate-900/90 rounded-2xl border border-slate-700 shadow-xl space-y-3">
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">
          <span className="px-2.5 py-1 rounded-md bg-emerald-950 border border-emerald-500/40 text-emerald-400 text-xs font-bold">
            제{selectedTeam + 1}국
          </span>
          <span className="text-xs text-slate-400">출석번호 {selectedSeat}번</span>
        </div>
        <div className="flex items-center gap-2 text-xs">
          <strong className="text-amber-400">{roomData.round} / 5 라운드</strong>
        </div>
      </div>
      <Gauges stats={myCountry.stats} money={myCountry.money} shield={myCountry.shield} />
    </div>
  );

  // 인트로: 선생님 화면에서 하루가 이야기하는 중
  if (roomData.phase === 'intro') {
    return (
      <div className="w-full max-w-2xl mx-auto p-3 md:p-6 space-y-4 text-white">
        <HaruCard mood="char_haru_normal" title="선생님 화면을 봐 줘!">
          지금 칠판(TV)에서 내가 2075년의 이야기를 들려주고 있어. 다 듣고 나면 우리 나라의 첫 투자를 정하게 될 거야.
        </HaruCard>
        {statusBar}
      </div>
    );
  }

  // 피해 도착: 2라운드 전 선택의 대가
  if (roomData.phase === 'round_damage') {
    const arrived = gameState.harms.filter(
      (h) => h.to === selectedTeam && h.arrived && h.arrivalRound === roomData.round
    );
    return (
      <div className="w-full max-w-2xl mx-auto p-3 md:p-6 space-y-4 text-white">
        <HaruCard
          mood={arrived.length ? 'char_haru_worried' : 'char_haru_hopeful'}
          title={arrived.length ? `${roomData.round}라운드 피해가 도착했어...` : `${roomData.round}라운드, 아직은 조용해!`}
          tone={arrived.length ? 'warn' : 'good'}
        >
          {arrived.length
            ? '2라운드 전에 내린 선택의 부작용이 지금 나타났어. 우리 나라는 어떻게 달라졌을까?'
            : roomData.round <= CONFIG.harmDelay
            ? '피해는 선택하고 2라운드 뒤에 도착해. 지금은 괜찮아 보여도 방심하면 안 돼!'
            : '이번 라운드에는 우리 나라에 도착한 피해가 없어.'}
        </HaruCard>

        {arrived.length > 0 && (() => {
          // 같은 사건은 한 장으로 묶는다 (여러 나라에서 번진 같은 피해가 줄줄이 나오지 않게)
          const groups = new Map<string, { ev: ReturnType<typeof findEvent>; froms: number[]; sum: Record<string, number> }>();
          for (const h of arrived) {
            const g = groups.get(h.eventId) ?? { ev: findEvent(CONFIG, h.eventId), froms: [], sum: {} };
            if (!g.froms.includes(h.from)) g.froms.push(h.from);
            for (const [k, v] of Object.entries(h.applied)) g.sum[k] = (g.sum[k] ?? 0) + v;
            groups.set(h.eventId, g);
          }
          const total: Record<string, number> = {};
          for (const g of groups.values()) for (const [k, v] of Object.entries(g.sum)) total[k] = (total[k] ?? 0) + v;
          const fmt = (m: Record<string, number>) =>
            Object.entries(m)
              .filter(([, v]) => Math.round(v) > 0)
              .map(([k, v]) => `${AXIS_LABEL[k] ?? k} -${Math.round(v)}`)
              .join(' · ');
          return (
            <>
              <div className="p-3 bg-rose-950/60 border border-rose-500/50 rounded-2xl text-center">
                <div className="text-[11px] text-rose-300">이번 라운드에 줄어든 점수</div>
                <div className="text-lg font-black text-rose-200">{fmt(total) || '거의 없어요'}</div>
              </div>
              {[...groups.values()].map((g, i) => {
                const others = g.froms.filter((f) => f !== selectedTeam).map((f) => `제${f + 1}국`);
                const mine = g.froms.includes(selectedTeam);
                return (
                  <div key={i} className="p-4 bg-rose-950/50 border border-rose-500/40 rounded-2xl space-y-2">
                    <div className="flex items-center gap-2 text-rose-300 font-bold text-sm">
                      <AlertTriangle className="w-5 h-5 text-rose-400 flex-shrink-0" />
                      <span className="break-keep">{g.ev?.title ?? '알 수 없는 피해'}</span>
                    </div>
                    <div className="text-[11px] font-bold text-amber-300">
                      {[mine ? '우리 나라의 선택' : '', others.length ? `${others.join('·')}에서 번졌어요` : ''].filter(Boolean).join(' + ')}
                    </div>
                    {g.ev?.desc && <p className="text-xs text-slate-300 leading-relaxed break-keep">{g.ev.desc}</p>}
                    <div className="text-sm font-black text-rose-300">{fmt(g.sum)}</div>
                  </div>
                );
              })}
            </>
          );
        })()}

        {statusBar}
        <div className="text-center text-xs text-slate-400 animate-pulse">
          선생님이 투표를 열 때까지 기다려 주세요.
        </div>
      </div>
    );
  }

  // 라운드 결과
  if (roomData.phase === 'round_result') {
    const dec = teamData?.lastDecision;
    return (
      <div className="w-full max-w-2xl mx-auto p-3 md:p-6 space-y-4 text-white">
        <HaruCard mood="char_haru_serious" title={`${roomData.round}라운드 결과가 나왔어!`}>
          우리 모둠이 다수결로 정한 선택이야. 혜택은 지금 오지만, 피해는 2라운드 뒤에 도착해.
        </HaruCard>

        <div className="p-4 bg-slate-900/90 rounded-2xl border border-slate-700 shadow-xl space-y-3">
          <h3 className="text-sm font-black text-white">우리 나라가 정한 투자</h3>
          {dec ? (
            <div className="grid grid-cols-2 gap-2">
              {CONFIG.domains.map((dom) => (
                <div key={dom.id} className="flex items-center gap-2 p-2.5 bg-slate-950/70 rounded-xl border border-slate-800">
                  <ChromaImg name={DOMAIN_ICON[dom.id]} alt="" className="w-9 h-9 object-contain" />
                  <div className="min-w-0">
                    <div className="text-[11px] text-slate-400">{dom.label}</div>
                    <div className="text-xs font-bold text-white break-keep">
                      {dec.levels[dom.id]}단계 · {dom.steps[dec.levels[dom.id]]?.name}
                    </div>
                  </div>
                </div>
              ))}
              <div className="flex items-center gap-2 p-2.5 bg-slate-950/70 rounded-xl border border-slate-800">
                <ChromaImg name={'icon_appro'} alt="" className="w-9 h-9 object-contain" />
                <div className="min-w-0">
                  <div className="text-[11px] text-slate-400">적정 기술</div>
                  <div className="text-xs font-bold text-white break-keep">
                    {dec.appropriate}단계 · {CONFIG.appropriate.steps[dec.appropriate]?.name}
                  </div>
                </div>
              </div>
            </div>
          ) : (
            <p className="text-xs text-slate-400">결과를 불러오는 중이에요...</p>
          )}

          <div className="flex items-center justify-between text-xs pt-1">
            <span className="flex items-center gap-1 text-slate-400">
              <Coins className="w-4 h-4 text-amber-400" /> 이번 라운드 자금 변화
            </span>
            <strong className={moneyDelta >= 0 ? 'text-emerald-400' : 'text-rose-400'}>
              {moneyDelta >= 0 ? '+' : ''}
              {Math.round(moneyDelta)}
            </strong>
          </div>
        </div>

        {teamData?.lastInvestigation && (
          <div className="p-4 bg-indigo-950/60 border border-indigo-500/40 rounded-2xl space-y-2">
            <div className="flex items-center gap-2 text-indigo-300 font-bold text-xs">
              <Eye className="w-4 h-4 text-indigo-400" />
              우리 모둠의 위험 조사 결과 (비밀 정보)
            </div>
            <p className="text-xs text-slate-300">
              미래 위험의 심각도: <strong className="text-amber-400">{teamData.lastInvestigation.severity}</strong> (영향 분야:{' '}
              {teamData.lastInvestigation.axes.map((a) => AXIS_LABEL[a] ?? a).join(', ') || '경미함'})
            </p>
          </div>
        )}

        {statusBar}
        <div className="text-center text-xs text-slate-400 animate-pulse">선생님이 다음 라운드를 시작할 때까지 기다려 주세요.</div>
      </div>
    );
  }

  // 투표
  return (
    <div className="w-full max-w-2xl mx-auto p-3 md:p-6 space-y-4 text-white pb-32">
      {statusBar}

      {isSubmitted ? (
        <HaruCard mood="char_haru_hopeful" title="투표 완료! 모둠원을 기다리는 중이야" tone="good">
          우리 모둠 {submittedCount} / {Math.max(seatedCount, submittedCount)}명이 제출했어. 선생님이 집계하면 결과가 나와.
        </HaruCard>
      ) : roomData.phase === 'revoting' ? (
        <HaruCard mood="char_haru_serious" title="표가 같아서 한 번 더 투표해!" tone="warn">
          모둠원과 의견을 맞춰 다시 골라 봐. 이미 제출했다면 선생님이 풀어 줘야 고칠 수 있어.
        </HaruCard>
      ) : (
        <HaruCard mood="char_haru_serious" title="우리 나라, 어디에 투자할까?">
          단계가 높을수록 시장 몫이 커지지만 돈이 많이 들고 실패할 수도 있어. 그리고 피해는 2라운드 뒤에 와!
        </HaruCard>
      )}

      {teamData?.lastInvestigation && (
        <div className="p-4 bg-indigo-950/60 border border-indigo-500/40 rounded-2xl space-y-2">
          <div className="flex items-center gap-2 text-indigo-300 font-bold text-xs">
            <Eye className="w-4 h-4 text-indigo-400" />
            우리 모둠의 위험 조사 결과 (비밀 정보)
          </div>
          <p className="text-xs text-slate-300">
            미래 위험의 심각도: <strong className="text-amber-400">{teamData.lastInvestigation.severity}</strong> (영향 분야:{' '}
            {teamData.lastInvestigation.axes.map((a) => AXIS_LABEL[a] ?? a).join(', ') || '경미함'})
          </p>
        </div>
      )}

      <div className="space-y-3">
        {CONFIG.domains.map((dom) => {
          const currentLevel = levels[dom.id];
          const lockedLevel = myLocks[dom.id] ?? 0;
          return (
            <div key={dom.id} className="p-4 bg-slate-900/90 rounded-2xl border border-slate-700 space-y-3">
              <div className="flex items-center gap-3">
                <ChromaImg name={DOMAIN_ICON[dom.id]} alt="" className="w-12 h-12 object-contain flex-shrink-0" />
                <div className="flex-1 min-w-0">
                  <div className="font-black text-base text-white">{dom.label}</div>
                  <div className="text-xs font-mono text-amber-400">
                    {currentLevel}단계 · {dom.cost[currentLevel]}원 · 성공 {Math.round((dom.success[currentLevel] ?? 1) * 100)}%
                  </div>
                </div>
              </div>

              <div className="grid grid-cols-5 gap-1.5">
                {dom.steps.map((_, sIdx) => {
                  const isLocked = sIdx < lockedLevel;
                  const isSelected = currentLevel === sIdx;
                  return (
                    <button
                      key={sIdx}
                      disabled={isLocked || isSubmitted}
                      onClick={() => setLevels({ ...levels, [dom.id]: sIdx })}
                      className={`py-3 rounded-xl text-xs font-bold flex flex-col items-center justify-center gap-0.5 transition-all cursor-pointer ${
                        isLocked
                          ? 'bg-slate-900/40 text-slate-600 border border-slate-900 cursor-not-allowed'
                          : isSelected
                          ? 'bg-emerald-600 text-white shadow-lg shadow-emerald-900/50 scale-[1.04]'
                          : 'bg-slate-800 hover:bg-slate-700 text-slate-300 border border-slate-700'
                      }`}
                    >
                      <span className="text-sm">{sIdx}</span>
                      <span className="text-[10px] opacity-80">{dom.cost[sIdx]}원</span>
                      {isLocked && <Lock className="w-3 h-3 text-red-500" />}
                    </button>
                  );
                })}
              </div>
              <div className="text-xs text-slate-300 leading-relaxed break-keep bg-slate-950/60 rounded-lg p-2.5">
                <strong className="text-white">{dom.steps[currentLevel]?.name}</strong>
                {dom.steps[currentLevel]?.desc ? ` — ${dom.steps[currentLevel].desc}` : ' — 투자하지 않아요'}
              </div>
            </div>
          );
        })}

        <div className="p-4 bg-slate-900/90 rounded-2xl border border-teal-500/30 space-y-3">
          <div className="flex items-center gap-3">
            <ChromaImg name={'icon_appro'} alt="" className="w-12 h-12 object-contain flex-shrink-0" />
            <div className="flex-1 min-w-0">
              <div className="font-black text-base text-teal-300">적정 기술</div>
              <div className="text-xs font-mono text-teal-400">
                {appropriate}단계 · {CONFIG.appropriate.cost[appropriate]}원 · 피해를 줄이고 이웃과 나눠요
              </div>
            </div>
          </div>
          <div className="grid grid-cols-4 gap-2">
            {CONFIG.appropriate.steps.map((_, aIdx) => (
              <button
                key={aIdx}
                disabled={isSubmitted}
                onClick={() => setAppropriate(aIdx)}
                className={`py-3 rounded-xl text-xs font-bold flex flex-col items-center gap-0.5 transition-all cursor-pointer ${
                  appropriate === aIdx
                    ? 'bg-teal-600 text-white shadow-lg shadow-teal-900/50 scale-[1.04]'
                    : 'bg-slate-800 hover:bg-slate-700 text-slate-300 border border-slate-700'
                }`}
              >
                <span className="text-sm">{aIdx}</span>
                <span className="text-[10px] opacity-80">{CONFIG.appropriate.cost[aIdx]}원</span>
              </button>
            ))}
          </div>
          <div className="text-xs text-slate-300 leading-relaxed break-keep bg-slate-950/60 rounded-lg p-2.5">
            <strong className="text-white">{CONFIG.appropriate.steps[appropriate]?.name}</strong>
            {CONFIG.appropriate.steps[appropriate]?.desc ? ` — ${CONFIG.appropriate.steps[appropriate].desc}` : ' — 투자하지 않아요'}
          </div>
        </div>
      </div>

      <div className="space-y-3">
        <div className="p-4 bg-slate-900/90 rounded-2xl border border-slate-700 space-y-2">
          <label className="flex items-center gap-3 text-sm font-bold text-slate-100 cursor-pointer">
            <input
              type="checkbox"
              disabled={isSubmitted}
              checked={wantInvestigate}
              onChange={(e) => {
                setWantInvestigate(e.target.checked);
                if (e.target.checked && !investigateChoice) setInvestigateChoice({ domain: 'bio', level: 3 });
              }}
              className="w-5 h-5 rounded"
            />
            <Search className="w-5 h-5 text-amber-400 flex-shrink-0" />
            <span className="break-keep">위험 사전 조사에 찬성해요 ({CONFIG.investigateCost}원)</span>
          </label>
          <p className="text-[11px] text-slate-400 pl-8">모둠 과반이 찬성하면 한 선택지의 미래 피해를 흐릿하게 미리 볼 수 있어요.</p>
          {wantInvestigate && (
            <div className="flex gap-2 text-sm pl-8">
              <select
                disabled={isSubmitted}
                value={investigateChoice?.domain || 'bio'}
                onChange={(e) =>
                  setInvestigateChoice({ domain: e.target.value as DomainId, level: investigateChoice?.level || 3 })
                }
                className="bg-slate-800 border border-slate-700 rounded-lg px-2 py-2 text-slate-200"
              >
                {CONFIG.domains.map((d) => (
                  <option key={d.id} value={d.id}>
                    {d.label}
                  </option>
                ))}
              </select>
              <select
                disabled={isSubmitted}
                value={investigateChoice?.level || 3}
                onChange={(e) =>
                  setInvestigateChoice({ domain: investigateChoice?.domain || 'bio', level: Number(e.target.value) })
                }
                className="bg-slate-800 border border-slate-700 rounded-lg px-2 py-2 text-slate-200"
              >
                {[1, 2, 3, 4].map((l) => (
                  <option key={l} value={l}>
                    {l}단계
                  </option>
                ))}
              </select>
            </div>
          )}
        </div>

        <div className="p-4 bg-slate-900/90 rounded-2xl border border-slate-700 space-y-2">
          <label className="flex items-center gap-3 text-sm font-bold text-slate-100 cursor-pointer">
            <input
              type="checkbox"
              disabled={isSubmitted}
              checked={wantRepair}
              onChange={(e) => setWantRepair(e.target.checked)}
              className="w-5 h-5 rounded"
            />
            <Wrench className="w-5 h-5 text-emerald-400 flex-shrink-0" />
            <span className="break-keep">도착한 피해를 복구하는 데 찬성해요</span>
          </label>
          <p className="text-[11px] text-slate-400 pl-8">
            과반이 찬성하면 망가진 사회·생태를 고쳐요. 비용은 개발비의 약 {CONFIG.repairCostMult}배예요.
          </p>
        </div>

        <div className="p-4 bg-amber-950/30 border border-amber-500/40 rounded-2xl space-y-2">
          <label className="block text-sm font-bold text-amber-300 break-keep">
            "이게 잘못되면 최악은 뭘까?" 한 줄로 적어 줘 (필수)
          </label>
          <textarea
            rows={2}
            disabled={isSubmitted}
            value={worstPrediction}
            onChange={(e) => setWorstPrediction(e.target.value)}
            placeholder="예: 편집된 유전자가 대를 이어 퍼져서 빈부격차가 굳어질 것이다."
            className="w-full px-3 py-2 bg-slate-950 border border-slate-700 rounded-lg text-sm text-white placeholder-slate-500 focus:outline-none focus:border-amber-400 resize-none"
          />
          <span className="text-[11px] text-slate-400 block">* 이 예측은 마지막 미래 세대 법정에서 우리의 책임을 따지는 증거가 돼요.</span>
        </div>

        {submitWarning && (
          <div className="p-3 bg-red-950/80 border border-red-500/50 rounded-xl text-xs text-red-300 font-bold">{submitWarning}</div>
        )}
      </div>

      {/* 하단 고정 바: 남은 돈 + 제출 */}
      <div className="fixed bottom-0 inset-x-0 z-30 bg-slate-950/95 backdrop-blur border-t border-slate-800 px-3 py-3">
        <div className="max-w-2xl mx-auto flex items-center gap-3">
          <div className="flex-1 min-w-0 text-xs leading-tight">
            <div className="flex items-center gap-1 text-slate-400">
              <Coins className="w-3.5 h-3.5 text-amber-400" /> 쓸 돈 <strong className="text-white">{spend}</strong> / 가진 돈{' '}
              <strong className="text-white">{Math.round(myCountry.money)}</strong>
            </div>
            <div className={remaining < 0 ? 'text-rose-400 font-bold' : 'text-emerald-400 font-bold'}>
              {remaining < 0 ? `${-remaining}원 모자라요! 일부 선택이 자동으로 낮아져요` : `남는 돈 ${remaining}`}
            </div>
            {countdown && (
              <div className="flex items-center gap-1 text-slate-400">
                <Timer className="w-3.5 h-3.5" /> 남은 시간 {countdown.label}
                {countdown.paused ? ' (멈춤)' : ''}
              </div>
            )}
          </div>
          <button
            onClick={handleSubmitBallot}
            disabled={isSubmitting || isSubmitted || !isVotingPhase}
            className="flex items-center gap-2 px-5 py-3 bg-emerald-600 hover:bg-emerald-500 disabled:opacity-40 text-white font-bold text-sm rounded-xl shadow-lg transition-colors cursor-pointer whitespace-nowrap"
          >
            {isSubmitted ? <CheckCircle className="w-4 h-4" /> : <Send className="w-4 h-4" />}
            <span>{isSubmitted ? '제출 완료' : '최종 제출'}</span>
          </button>
        </div>
      </div>
    </div>
  );
};
````

````
```tsx
import React, { useState, useEffect } from 'react';
import { court, VERDICT_NAMES, findEvent } from '../../game/engine.ts';
import type { GameState } from '../../game/engine.ts';
import { assetUrl } from '../../game/assets.ts';
import { Scale, Gavel, ArrowLeft, ArrowRight, ShieldCheck, AlertCircle, FileText, Sparkles } from 'lucide-react';
import confetti from 'canvas-confetti';

interface CourtViewProps {
  gameState: GameState;
  isTeacher?: boolean;
  onFinish?: () => void;
  teamInputsMap?: Record<number, Record<number, { ballot: { worst: string } }>>;
  initialCountry?: number;
}

export const CourtView: React.FC<CourtViewProps> = ({
  gameState,
  isTeacher = false,
  onFinish,
  teamInputsMap = {},
  initialCountry = 0,
}) => {
  const [currentCountry, setCurrentCountry] = useState(initialCountry);
  const [showFinalQuote, setShowFinalQuote] = useState(false);

  const judgements = court(gameState);
  const cData = gameState.countries[currentCountry];
  const judgement = judgements[currentCountry];

  useEffect(() => {
    if (judgement.verdicts.includes('clear')) {
      confetti({
        particleCount: 60,
        spread: 70,
        origin: { y: 0.6 },
      });
    }
  }, [currentCountry]);

  // Find harms caused by this country
  const causedHarms = gameState.harms.filter((h) => h.from === currentCountry);
  const causedEvents = causedHarms
    .map((h) => findEvent(gameState.cfg, h.eventId))
    .filter(Boolean);
  // 같은 사건이 여러 번 일어났으면 한 줄로 묶고 횟수를 센다
  const causedGroups: { ev: NonNullable<(typeof causedEvents)[number]>; count: number }[] = [];
  for (const ev of causedEvents) {
    if (!ev) continue;
    const g = causedGroups.find((x) => x.ev.id === ev.id);
    if (g) g.count += 1;
    else causedGroups.push({ ev, count: 1 });
  }

  // Student predictions recorded by this team
  const teamInputs = teamInputsMap[currentCountry] || {};
  const predictions = Object.values(teamInputs)
    .map((inp) => inp.ballot?.worst)
    .filter(Boolean);

  return (
    <div className="w-full max-w-5xl mx-auto p-4 md:p-8 space-y-6 animate-fade-in text-white">
      {/* Court Header */}
      <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-3 p-4 bg-slate-900/90 rounded-2xl border border-slate-700 shadow-xl">
        <div className="flex items-center gap-3">
          <div className="p-2.5 rounded-xl bg-amber-500/20 text-amber-400 border border-amber-500/30">
            <Gavel className="w-6 h-6" />
          </div>
          <div>
            <h2 className="text-xl md:text-2xl font-black text-amber-300">
              미래 세대 법정 (Future Generation Court)
            </h2>
            <p className="text-xs text-slate-400">
              2075년 미래의 아이들이 2025년 조상들의 선택을 직접 심판합니다.
            </p>
          </div>
        </div>

        {/* Country Selector Tabs */}
        <div className="flex flex-wrap items-center gap-1.5">
          {gameState.countries.map((_, idx) => (
            <button
              key={idx}
              onClick={() => {
                setCurrentCountry(idx);
                setShowFinalQuote(false);
              }}
              className={`px-3 py-1.5 rounded-lg text-xs font-bold transition-all cursor-pointer ${
                currentCountry === idx && !showFinalQuote
                  ? 'bg-amber-500 text-slate-950 font-black shadow-lg scale-105'
                  : 'bg-slate-800 text-slate-400 hover:text-white border border-slate-700'
              }`}
            >
              제{idx + 1}국
            </button>
          ))}
          <button
            onClick={() => setShowFinalQuote(true)}
            className={`px-3 py-1.5 rounded-lg text-xs font-bold transition-all cursor-pointer ${
              showFinalQuote
                ? 'bg-emerald-500 text-slate-950 font-black shadow'
                : 'bg-slate-800 text-slate-300 hover:text-white border border-slate-700'
            }`}
          >
            📜 요나스의 한 줄
          </button>
        </div>
      </div>

      {!showFinalQuote ? (
        <div className="space-y-6">
          {/* Main Verdict Card */}
          <div className="relative rounded-3xl overflow-hidden border-2 border-amber-500/40 bg-slate-950 shadow-2xl">
            {/* Background Courtroom Image */}
            <img
              src={assetUrl('court_bg')}
              alt="법정 배경"
              className="absolute inset-0 w-full h-full object-cover opacity-20"
              onError={(e) => {
                (e.currentTarget as HTMLImageElement).style.display = 'none';
              }}
            />
            <div className="relative p-6 md:p-8 space-y-6">
              {/* Country Badge & Verdict Title */}
              <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 border-b border-slate-800 pb-6">
                <div>
                  <span className="text-xs font-bold text-amber-400 tracking-widest uppercase">
                    국가별 책임 판결문
                  </span>
                  <h3 className="text-2xl md:text-4xl font-black text-white mt-1">
                    제{currentCountry + 1}국에 대한 판결
                  </h3>
                </div>

                <div className="flex flex-wrap items-center gap-2">
                  {judgement.mercy && (
                    <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-emerald-950/90 border border-emerald-500/50 text-emerald-300 text-xs font-bold">
                      <ShieldCheck className="w-4 h-4 text-emerald-400" />
                      참작 및 감형 사유 인정
                    </div>
                  )}
                  <div className="px-3 py-1.5 rounded-full bg-slate-800 border border-slate-700 text-xs font-bold text-amber-400">
                    보유 자금: {Math.round(cData.money)} / 생존 지표: {Math.round(cData.stats.health + cData.stats.ecology + cData.stats.society)}
                  </div>
                </div>
              </div>

              {/* Verdict Names & Artwork Badges */}
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                {judgement.verdicts.map((vKey) => (
                  <div
                    key={vKey}
                    className={`flex items-center gap-4 p-4 rounded-2xl border ${
                      vKey === 'clear'
                        ? 'bg-emerald-950/60 border-emerald-500/50 text-emerald-200'
                        : 'bg-rose-950/50 border-rose-500/40 text-rose-200'
                    }`}
                  >
                    <div className="w-16 h-16 rounded-xl overflow-hidden bg-black/50 border border-white/10 flex-shrink-0">
                      <img
                        src={assetUrl(`icon_verdict_${vKey}` as any)}
                        alt={VERDICT_NAMES[vKey]}
                        className="w-full h-full object-cover"
                        onError={(e) => {
                          (e.currentTarget as HTMLImageElement).src = assetUrl('icon_investigate');
                        }}
                      />
                    </div>
                    <div>
                      <div className="text-[11px] font-bold uppercase tracking-wider opacity-75">
                        {vKey === 'clear' ? '칭찬 판결' : '죄목 선고'}
                      </div>
                      <div className="text-base font-bold text-white mt-0.5 leading-snug">
                        {VERDICT_NAMES[vKey]}
                      </div>
                    </div>
                  </div>
                ))}
              </div>

              {/* Evidence Section */}
              <div className="grid grid-cols-1 md:grid-cols-2 gap-6 pt-2">
                {/* Evidence 1: Predictions vs Reality */}
                <div className="p-5 bg-slate-900/90 rounded-2xl border border-slate-800 space-y-3">
                  <h4 className="text-sm font-bold text-slate-300 flex items-center gap-2">
                    <FileText className="w-4 h-4 text-amber-400" />
                    증거 ① 우리가 적은 최악의 예측 vs 실제로 일어난 일
                  </h4>

                  <div className="space-y-2 text-xs">
                    <div className="p-3 bg-slate-950 rounded-xl border border-slate-800">
                      <span className="font-bold text-amber-400 block mb-1">
                        우리가 투표 때 적은 최악의 시나리오:
                      </span>
                      {predictions.length > 0 ? (
                        <ul className="list-disc list-inside space-y-1 text-slate-300">
                          {predictions.slice(0, 3).map((p, i) => (
                            <li key={i} className="italic">"{p}"</li>
                          ))}
                        </ul>
                      ) : (
                        <p className="text-slate-500 italic">기록된 예측 없음</p>
                      )}
                    </div>

                    <div className="p-3 bg-slate-950 rounded-xl border border-slate-800">
                      <span className="font-bold text-rose-400 block mb-1">
                        실제로 발생한 피해 사건:
                      </span>
                      {causedGroups.length > 0 ? (
                        <div className="space-y-1.5">
                          {causedGroups.map(({ ev, count }, i) => (
                            <div key={i} className="text-slate-300">
                              <span className="font-semibold text-white">{ev.title}</span>
                              {count > 1 && <span className="ml-1 text-rose-300 font-bold">×{count}회</span>}
                              <span>: {ev.desc}</span>
                              <div className="mt-0.5 text-[10px] text-amber-300">
                                [{ev.kind === 'real' ? '실제로 있었던 일' : '전문가들이 경고하는 가정'}]
                              </div>
                            </div>
                          ))}
                        </div>
                      ) : (
                        <p className="text-emerald-400">발생한 중대 피해 사건 없음</p>
                      )}
                    </div>
                  </div>
                </div>

                {/* Evidence 2: Numerical Records */}
                <div className="p-5 bg-slate-900/90 rounded-2xl border border-slate-800 space-y-3">
                  <h4 className="text-sm font-bold text-slate-300 flex items-center gap-2">
                    <Scale className="w-4 h-4 text-emerald-400" />
                    증거 ② 윤리적 노력 및 방치 기록
                  </h4>

                  <div className="grid grid-cols-2 gap-3 text-xs">
                    <div className="p-3 bg-slate-950 rounded-xl border border-slate-800">
                      <span className="text-slate-400 block">위험 조사 횟수:</span>
                      <span className="text-lg font-black text-white">{cData.investigations}회</span>
                      <p className="text-[10px] text-slate-500 mt-0.5">미리 위험을 알아채려 했는가?</p>
                    </div>

                    <div className="p-3 bg-slate-950 rounded-xl border border-slate-800">

                      <span className="text-lg font-black text-rose-400">{cData.ignoredWarnings}회</span>
                      <p className="text-[10px] text-slate-500 mt-0.5">모둠원의 반대로 묵살된 경고</p>
                    </div>

                    <div className="p-3 bg-slate-950 rounded-xl border border-slate-800">
                      <span className="text-slate-400 block">적정 기술 투자액:</span>
                      <span className="text-lg font-black text-teal-400">{cData.approSpent}원</span>

                      <span className="text-slate-400 block">피해 복구 횟수:</span>

                <span>최종 마무리</span>
                <Sparkles className="w-4 h-4 text-slate-950" />

            <h3 className="text-xl md:text-2xl font-bold text-indigo-300 tracking-wide uppercase">
              한스 요나스 (Hans Jonas) — 책임의 정언명령
````

# ③ 확인 체크리스트

1. 주소 뒤에 아무것도 안 붙이고 열기 → 학생 입장(방 코드 입력) 화면이 보여야 합니다.
2. `주소/?teacher` 로 열기 → 교사 화면(수업 기록 보관함)이 보여야 합니다.
3. 교사 화면에서 새 수업 만들기 (방 생성은 약 20초 걸릴 수 있어요) → 방 코드 확인.
4. 다른 기기(폰)에서 `주소/?code=방코드` 로 열기 → 코드가 미리 채워져 있어야 합니다.
5. 모둠, 번호 선택 → 대기 화면에서 하루 이미지가 초록 배경 없이 보여야 합니다.
6. 교사: 게임 시작 → 다음 진행으로 투표까지 → 학생 폰에서 제출 → 교사 "투표 집계 마감" → 결과 화면까지 넘어가야 합니다.
7. 동률이 나면 학생 폰에서 투표 폼이 다시 열려야 합니다.

# ④ Firebase 콘솔 규칙 수정 (맨 마지막에, 위 코드 적용·확인 후)

**주의:** 아래 규칙은 테스트하지 못했습니다. 게시 전에 반드시 규칙 플레이그라운드에서 확인하고, 현재 규칙은 메모장에 백업해 두세요.

## 1) 익명 로그인 켜기 (규칙 적용 전에 먼저)
1. console.firebase.google.com → 프로젝트 `arctic-computing-sfs6l`
2. 왼쪽 **빌드 → Authentication → Sign-in method → 익명** 사용 설정
3. 앱을 새로 열어 학생 접속 시 400 오류(`identitytoolkit ... signUp`)가 사라졌는지 확인

## 2) 규칙 편집창 열기
1. **빌드 → Firestore Database → 규칙** 탭
2. 이 앱은 `(default)`가 아니라 **`ai-studio-2075earth2075-…`** 이름의 데이터베이스를 씁니다. 상단 DB 선택에서 그 DB를 고르세요.
3. 현재 규칙 전체를 복사해 백업

## 3) 새 규칙 붙여넣기

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isTeacher() {
      return signedIn()
        && request.auth.token.email != null
        && request.auth.token.email_verified == true
        && (request.auth.token.email.matches('.*@jeonnong[.]ms[.]kr')
            || request.auth.token.email == 'sangwoo59590@gmail.com');
    }

    match /rooms/{roomId} {
      allow read: if true;
      allow create, update, delete: if isTeacher();

      match /teams/{teamId} {
        allow read: if true;
        allow create, delete: if isTeacher();
        // 학생은 좌석(seats)만 바꿀 수 있음
        allow update: if isTeacher() ||
          (signedIn() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['seats']));

        match /inputs/{seatId} {
          allow read: if true;
          allow create, update: if signedIn();
          allow delete: if isTeacher();
        }
      }

      match /{sub}/{docId} {  // backups, logs
        allow read, write: if isTeacher();
      }
    }
  }
}
```

## 4) 게시 전 플레이그라운드 테스트
- 비로그인으로 `rooms/아무값` **삭제** → 거부되어야 함
- 익명 로그인으로 `rooms/x/teams/0/inputs/5` **쓰기** → 허용, `rooms/x` **삭제** → 거부
- 기대대로면 **게시**

## 5) 게시 후 확인
- 학생 폰: 입장 → 좌석 선택 → 투표 제출이 되는지
- 교사: **구글 로그인** 후 방 만들기가 되는지
- 하나라도 안 되면 백업한 규칙으로 되돌리기

## 이 규칙의 영향
- 교사도 **구글 로그인**을 해야 방 생성·진행·삭제가 됩니다 (`/?teacher`만으로는 안 됨).
- 허용 교사 계정: `@jeonnong.ms.kr` 전체, `sangwoo59590@gmail.com` (다른 계정이면 `isTeacher()` 수정)
- 학생이 다른 학생의 표를 덮어쓰는 것까지는 못 막습니다 (현재 구조의 한계).
- 구글 로그인 팝업이 안 뜨면 **Authentication → 설정 → 승인된 도메인**에 `service-2075-earth-2075.ai.studio` 추가
